Callbacks

ℹ️
La convención Node.js usa error-first callbacks: el primer argumento es siempre el error, el segundo el resultado.
callbacks.js
// Patrón error-first
function leerDatos(id, callback) {
  db.find(id, (err, resultado) => {
    if (err) return callback(err);
    callback(null, resultado);
  });
}

leerDatos(1, (err, data) => {
  if (err) {
    console.error('Error:', err.message);
    return;
  }
  console.log(data);
});

// Crear Promise
const promesa = new Promise((resolve, reject) => {
  if (condicion) resolve('éxito');
  else reject(new Error('falló'));

// Métodos estáticos de Promise
Promise.resolve('valor');      // siempre resuelve
Promise.reject(new Error());  // siempre rechaza

// Esperar todas (falla si una falla)
const [a, b, c] = await Promise.all([p1, p2, p3]);

// Esperar todas sin importar fallos
const resultados = await Promise.allSettled([p1, p2, p3]);
resultados.forEach(r => {
  if (r.status === 'fulfilled') console.log(r.value);
  else console.error(r.reason);
});

// La primera que resuelva
const primera = await Promise.race([p1, p2, p3]);

// Primera que resuelva con éxito
const cualquiera = await Promise.any([p1, p2, p3]);

cluster / worker_threads
Concurrencia y paralelismo en Node.js
cluster.js
const cluster = require('cluster');
const os = require('os');

if (cluster.isPrimary) {
  const numCPUs = os.cpus().length;
  
  for (let i = 0; i < numCPUs; i++) {
    cluster.fork();
  }
  
  cluster.on('exit', (worker, code) => {
    console.log(`Worker ${worker.process.pid} murió`);
    cluster.fork(); // reiniciar
  });
} else {
  // Cada worker ejecuta el servidor
  require('./server');
}
worker.js
const { Worker, isMainThread,
        parentPort, workerData } = require('worker_threads');

if (isMainThread) {
  const worker = new Worker(__filename, {
    workerData: { datos: [1, 2, 3] }
  });
  worker.on('message', msg => console.log(msg));
  worker.on('error', err => console.error(err));
  worker.on('exit', code => console.log('Exit:', code));
} else {
  // Tarea intensiva en CPU
  const result = workerData.datos.reduce((a, b) => a + b);
  parentPort.postMessage({ resultado: result });
}

Manejo de Errores
Patrones y mejores prácticas para errores
⚠️
Nunca dejes un unhandledRejection o uncaughtException sin manejar en producción — puede derribar el proceso.
error-handling.js
// Error personalizado
class AppError extends Error {
  constructor(message, statusCode = 500) {
    super(message);
    this.name = this.constructor.name;
    this.statusCode = statusCode;
    Error.captureStackTrace(this, this.constructor);
  }
}

throw new AppError('No encontrado', 404);

// Try/catch en async
async function handler(req, res, next) {
  try {
    const data = await buscarDatos(req.params.id);
    res.json(data);
  } catch (err) {
    next(err); // pasar a middleware de error
  }
}

// Wrapper para rutas async (evitar repetir try/catch)
const asyncHandler = fn => (req, res, next) =>
  Promise.resolve(fn(req, res, next)).catch(next);

app.get('/ruta', asyncHandler(async (req, res) => {
  const data = await obtenerData();
  res.json(data);
}));

// Errores no capturados
process.on('uncaughtException', (err) => {
  console.error('Error no capturado:', err);
  process.exit(1);
});

process.on('unhandledRejection', (reason, promise) => {
  console.error('Promise rechazada:', reason);
  process.exit(1);
});
